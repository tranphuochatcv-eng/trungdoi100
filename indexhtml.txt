                movieId: st.movieId,
                showtimeId: st.id,
                selectedSeats: [],
                concessions: {},
                paymentMethod: 'Momo',
                customerName: '',
                customerPhone: '',

            renderBookingStep2(); // Go straight to seats if showtime chosen

        function updateStepperUI(step) {
            for (let i = 1; i <= 4; i++) {
                const ind = document.getElementById(`step-ind-${i}`);
                const span = ind.querySelector('span');
                if (i < step) {
                    ind.className = 'flex items-center gap-2 text-emerald-400 font-semibold';
                    span.className = 'w-6 h-6 rounded-full bg-emerald-600 text-white flex items-center justify-center font-bold';
                    span.innerHTML = '<i class="fa-solid fa-check text-xs"></i>';
                } else if (i === step) {
                    ind.className = 'flex items-center gap-2 text-cinema-red font-bold';
                    span.className = 'w-6 h-6 rounded-full bg-cinema-red text-white flex items-center justify-center font-bold';
                    span.innerText = i;
                } else {
                    ind.className = 'flex items-center gap-2 text-slate-500';
                    span.className = 'w-6 h-6 rounded-full bg-cinema-border text-slate-300 flex items-center justify-center font-bold';
                    span.innerText = i;
                }
            }
        }

        function toggleSeat(seatNum) {
            const idx = bookingState.selectedSeats.indexOf(seatNum);
            if (idx > -1) {
                bookingState.selectedSeats.splice(idx, 1);
            } else {
                if (bookingState.selectedSeats.length >= 8) {
                    showToast('Bạn chỉ có thể đặt tối đa 8 ghế trong 1 lần giao dịch.', 'error');
                    return;
                }
                bookingState.selectedSeats.push(seatNum);
            }
            renderBookingStep2();
        }

            renderBookingStep3();

        function updateConcessionQty(id, delta) {
            const current = bookingState.concessions[id] || 0;
            const updated = current + delta;
            if (updated <= 0) {
                delete bookingState.concessions[id];
            } else {
                bookingState.concessions[id] = updated;
            }
            renderBookingStep3();
        }

        function confirmCheckout(grandTotal) {
            const name = document.getElementById('checkout-name').value.trim();
            const phone = document.getElementById('checkout-phone').value.trim();
            const email = document.getElementById('checkout-email').value.trim();

            if (!name || !phone) {
                showToast('Vui lòng nhập đầy đủ họ tên và số điện thoại.', 'error');
                return;
            }

            const showtimes = getShowtimes();
            const movies = getMovies();
            const st = showtimes.find(s => s.id === bookingState.showtimeId);
            const movie = movies.find(m => m.id === bookingState.movieId);

            let concessionsList = [];
            for (const [id, qty] of Object.entries(bookingState.concessions)) {
                const c = defaultConcessions.find(item => item.id === id);
                if (c && qty > 0) {
                    concessionsList.push({name: c.name, qty, price: c.price * qty});
                }
            }

            const newBooking = {
                id: 'CM' + Math.floor(10000 + Math.random() * 90000),
                customerName: name,
                phone: phone,
                email: email,
                movieTitle: movie.title,
                cinema: st.cinema,
                room: st.room,
                time: st.time,
                seats: [...bookingState.selectedSeats],
                concessions: concessionsList,
                total: grandTotal,
                paymentMethod: bookingState.paymentMethod,
                date: new Date().toLocaleString()
            };

            const bookings = getBookings();
            bookings.unshift(newBooking);
            localStorage.setItem('cinemax_bookings', JSON.stringify(bookings));

            closeBookingModal();
            showToast('Đặt vé thành công! Vé đã được lưu vào mục "Vé Của Tôi".', 'success');
            updateMyTicketsBadge();
            openTicketSuccessModal(newBooking);
        }

        function handleMovieFormSubmit(e) {
            e.preventDefault();
            const id = document.getElementById('form-movie-id').value;
            const title = document.getElementById('form-movie-title').value;
            const genre = document.getElementById('form-movie-genre').value;
            const duration = parseInt(document.getElementById('form-movie-duration').value);
            const status = document.getElementById('form-movie-status').value;
            const rating = document.getElementById('form-movie-rating').value;
            const poster = document.getElementById('form-movie-poster').value;
            const trailer = document.getElementById('form-movie-trailer').value;
            const description = document.getElementById('form-movie-desc').value;

            let movies = getMovies();
            if (id) {
                // Edit
                movies = movies.map(m => m.id == id ? {...m, title, genre, duration, status, rating, poster, trailer, description} : m);
                showToast('Cập nhật thông tin phim thành công!', 'success');
            } else {
                // Add
                const newId = movies.length > 0 ? Math.max(...movies.map(m => m.id)) + 1 : 1;
                movies.push({id: newId, title, genre, duration, status, rating, poster, trailer, description});
                showToast('Thêm phim mới thành công!', 'success');
            }

            localStorage.setItem('cinemax_movies', JSON.stringify(movies));
            closeMovieFormModal();
            renderAdminMovies();
            renderAdminStats();
        }

        // Charts Initialization
        function initCharts() {
            const ctxMovie = document.getElementById('revenueByMovieChart').getContext('2d');
            const ctxDay = document.getElementById('revenueByDayChart').getContext('2d');

            const movies = getMovies();
            const bookings = getBookings();

            // Calculate revenue by movie
            const movieRevenue = {};
            movies.forEach(m => movieRevenue[m.title] = 0);
            bookings.forEach(b => {
                if (movieRevenue[b.movieTitle] !== undefined) {
                    movieRevenue[b.movieTitle] += b.total;
                } else {
                    movieRevenue[b.movieTitle] = b.total;
                }
            });

            chartsInstance.movieChart = new Chart(ctxMovie, {
                type: 'doughnut',
                data: {
                    labels: Object.keys(movieRevenue),
                    datasets: [{
                        data: Object.values(movieRevenue),
                        backgroundColor: ['#E50914', '#F59E0B', '#3B82F6', '#8B5CF6', '#10B981']
                    }]
                },
                options: {
                    responsive: true,
                    plugins: {
                        legend: { position: 'bottom', labels: { color: '#94A3B8', font: { size: 11 } } }
                    }
                }
            });

            chartsInstance.dayChart = new Chart(ctxDay, {
                type: 'bar',
                data: {
                    labels: ['Thứ 2', 'Thứ 3', 'Thứ 4', 'Thứ 5', 'Thứ 6', 'Thứ 7', 'Chủ Nhật'],

        function updateCharts() {
            if (chartsInstance.movieChart && chartsInstance.dayChart) {
                const movies = getMovies();
                const bookings = getBookings();
                const movieRevenue = {};
                movies.forEach(m => movieRevenue[m.title] = 0);
                bookings.forEach(b => {
                    if (movieRevenue[b.movieTitle] !== undefined) {
                        movieRevenue[b.movieTitle] += b.total;
                    } else {
                        movieRevenue[b.movieTitle] = b.total;
                    }
                });

                chartsInstance.movieChart.data.labels = Object.keys(movieRevenue);
                chartsInstance.movieChart.data.datasets[0].data = Object.values(movieRevenue);
                chartsInstance.movieChart.update();
            }
        }
